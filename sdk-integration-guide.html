Groups

Define local AdMob groups

object PublisherAdMobConfigProvider {
    fun build() = AdMobConfig(
        interstitialGroups = mapOf(
            AdMobConfig.DEFAULT_GROUP_REF to AdMobGroupConfig(
                unitIds = listOf("ca-app-pub-xxx/111"),
                preloadOnInit = true
            ),
            "interstitial_level_end" to AdMobGroupConfig(
                unitIds = listOf("ca-app-pub-xxx/112"),
                allowFallbackToDefault = true
            )
        ),
        rewardedGroups = mapOf(
            "rewarded_bonus_claim" to AdMobGroupConfig(
                unitIds = listOf("ca-app-pub-xxx/222"),
                preloadOnInit = true,
                allowFallbackToDefault = true
            )
        ),
        bannerGroups = mapOf(
            "banner_content" to AdMobGroupConfig(
                unitIds = listOf("ca-app-pub-xxx/333")
            )
        ),
        nativeGroups = mapOf(
            "native_feed_card" to AdMobGroupConfig(
                unitIds = listOf("ca-app-pub-xxx/444")
            )
        ),
        appOpenGroups = mapOf(
            "app_open_cold_start" to AdMobGroupConfig(
                unitIds = listOf("ca-app-pub-xxx/555"),
                preloadOnInit = true
            )
        )
    )
}

Manual Load

Load interstitial in your app layer

val request = Monetization.preloadAdMobInterstitial("interstitial_level_end")
if (request != null) {
    InterstitialAd.load(
        context,
        request.unitId,
        AdRequest.Builder().build(),
        object : InterstitialAdLoadCallback() {
            override fun onAdLoaded(ad: InterstitialAd) {
                Monetization.markAdMobInterstitialLoaded(request.groupRef, ad)
            }

            override fun onAdFailedToLoad(error: LoadAdError) {
                Monetization.markAdMobInterstitialLoadFailed(request.groupRef)
            }
        }
    )
}

Install

Add the React Native AdMob wrapper

npm install @amos-sdk/react-native-admob@1.0.19

Provider

Configure AMOS at app root

import { AmosProvider } from '@amos-sdk/react-native-admob';

const amosConfig = {
  apiKey: 'YOUR_API_KEY',
  appId: 'YOUR_APP_ID',
  userId: 'publisher-user-123',
  bootstrap: {
    firstOpenEvent: 'first_open',
    metrics: ['platform', 'country', 'install_source', 'device_model'],
    installSourceFallback: 'other',
  },
  admob: {
    interstitialGroups: {
      default: { unitIds: ['ca-app-pub-xxx/111'] },
    },
    rewardedGroups: {
      default: { unitIds: ['ca-app-pub-xxx/222'], preloadOnInit: true },
    },
    bannerGroups: {
      default: { unitIds: ['ca-app-pub-xxx/333'] },
    },
  },
};

export function App() {
  return (
    <AmosProvider config={amosConfig}>
      <RootNavigator />
    </AmosProvider>
  );
}

Usage

Use wrapper methods directly

import { AmosBannerAd, useAmos } from '@amos-sdk/react-native-admob';

function RewardScreen() {
  const amos = useAmos();

  async function onRewardTap() {
    const rewarded = await amos.showRewarded('default');
    if (rewarded) {
      await amos.trackEvent('rewarded_ad_complete');
      grantCreditToUser();
    }
  }

  return <AmosBannerAd groupRef="default" />;
}

Ads

Load, show, and render safely

final ads = state.decision.ads;

if (state.decision.provider == 'admob' && ads.interstitialEnabled) {
  final load = await AmosAdMob.instance.preloadInterstitial();
  if (load.loaded) await AmosAdMob.instance.showInterstitial();
}

if (ads.rewardedEnabled) {
  await AmosAdMob.instance.preloadRewarded();
  final result = await AmosAdMob.instance.showRewarded();
  if (result.rewardEarned) grantReward();
}

// Widget tree:
return ads.bannerEnabled
    ? const AmosBannerAd()
    : const SizedBox.shrink();

Production checklist

Before release

  • Confirm `first_open`, events, and bootstrap metrics appear in AMOS logs.
  • Validate rewarded reward amount and app-side credit capping.
  • Use stable user IDs for repeatable segmentation and history.
  • Keep publisher reward grants in app code, not in SDK-only code.
  • Test app fallback before enabling AMOS-controlled ad serving broadly.
