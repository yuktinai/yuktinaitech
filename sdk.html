Serving control

Keep rollout safe.

  • Use AMOS for events, metrics, bootstrap, and decisions in every integration mode.
  • Use AMOS-controlled AdMob serving only when the publisher app is ready.
  • Set `amos_ad_serving_enabled=false` remotely to fall back to the app's local ad loading path.

Install

Kotlin dependency setup

dependencyResolutionManagement {
    repositories {
        google()
        mavenCentral()
    }
}

dependencies {
    // AMOS controls AdMob load/show helpers
    implementation("com.yuktinai:amos-sdk-admob:1.0.10")

    // Or core-only/manual serving
    // implementation("com.yuktinai:amos-sdk:1.0.10")
}

Initialize

Start AMOS early in app startup

Monetization.init(
    context = applicationContext,
    apiKey = "YOUR_API_KEY",
    appId = "YOUR_APP_ID",
    userId = "publisher-user-123"
) {
    AmosAdMob.configure(PublisherAdMobConfigProvider.build())
    AmosAdMob.initialize(this)
}

Use

Track and show from app surfaces

Monetization.trackEventIfAllowed("image_selection")
Monetization.trackMetricIfAllowed("device_model", Build.MODEL)

AmosAdMob.showRewarded(
    activity = this,
    groupRef = "rewarded_bonus_claim",
    onRewardEarned = { grantCreditToUser() }
)

AmosAdMob.showInterstitial(
    activity = this,
    groupRef = "interstitial_level_end"

Install

React Native package setup

npm install @amos-sdk/react-native@2.0.6

# AMOS + AdMob wrapper hooks/components
npm install @amos-sdk/react-native-admob@1.0.13

Provider

Initialize once near app root

import { AmosProvider } from '@amos-sdk/react-native-admob';

const amosConfig = {
  apiKey: 'YOUR_API_KEY',
  appId: 'YOUR_APP_ID',
  userId: 'publisher-user-123',
  bootstrap: {
    firstOpenEvent: 'first_open',
    metrics: ['platform', 'country', 'install_source', 'device_model'],
    installSourceFallback: 'other',
  },
  admob: {
    rewardedGroups: {
      default: { unitIds: ['ca-app-pub-xxx/222'], preloadOnInit: true },
    },
    bannerGroups: {
      default: { unitIds: ['ca-app-pub-xxx/333'] },
    },
  },
};

export default function App() {
  return (
    <AmosProvider config={amosConfig}>
      <RootNavigator />
    </AmosProvider>
  );
}

Use

Track events and call ad helpers

import { AmosBannerAd, useAmos } from '@amos-sdk/react-native-admob';

function RewardEntry() {
  const amos = useAmos();

  async function onRewardTap() {
    const rewarded = await amos.showRewarded('default');
    if (rewarded) {
      await amos.trackEvent('rewarded_ad_complete');
      grantCreditToUser();
    }
  }

  return <AmosBannerAd groupRef="default" />;
}

Recommended rollout

Keep first integration focused.

  • Start with default ad groups first.
  • Validate one rewarded flow before enabling AMOS serving broadly.
  • Keep reward grants and final UX decisions in publisher app code.
  • Use `amos_ad_serving_enabled=false` if a live app needs immediate fallback.
